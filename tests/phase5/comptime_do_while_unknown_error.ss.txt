function negTest(n: int) {
    comptime {
        do {
            // body 留空 — 不影响触发路径,cond `n` 评估时 n 为非 ct func 参数,
            // evalIdent fallthrough → mv-runtime → isCt == 0 → do-while loud error
        } while (n)
    }
}

function main() {
    negTest(5)
}
