# 58 - i18n & l10n (国际化与本地化)

## 设计理念

> 多语言支持内置到官方包。消息文件 + 注解注入，零硬编码字符串。

## 消息文件

```yaml
# resources/i18n/messages.yml (默认语言)
greeting: "Hello, {name}!"
user:
  notFound: "User not found"
  created: "User {name} created successfully"
  deleted: "User {id} has been deleted"
error:
  unauthorized: "Please login first"
  forbidden: "Access denied"
  validation: "{field} is invalid"
```

```yaml
# resources/i18n/messages_zh.yml
greeting: "你好，{name}！"
user:
  notFound: "用户不存在"
  created: "用户 {name} 创建成功"
  deleted: "用户 {id} 已删除"
error:
  unauthorized: "请先登录"
  forbidden: "没有权限"
  validation: "{field} 不合法"
```

```yaml
# resources/i18n/messages_ja.yml
greeting: "こんにちは、{name}！"
user:
  notFound: "ユーザーが見つかりません"
```

## 使用

```simplescript
import { I18n, Message } from "ss/i18n"

@Service
class UserService(i18n: I18n) {

    function greet(name: string, locale: string): string {
        return i18n.get("greeting", locale, Map.of(["name", name]))
        // locale="zh" → "你好，Alice！"
        // locale="en" → "Hello, Alice!"
    }

    function deleteUser(id: long, locale: string): Result<string, Error> {
        db.delete<User>(id)?
        return Result.Ok(i18n.get("user.deleted", locale, Map.of(["id", id])))
    }
}
```

## Web 集成

```simplescript
import { I18n, Locale } from "ss/i18n"
import { RestController, GetMapping } from "ss/web"

@RestController
class ApiController(i18n: I18n) {

    @GetMapping("/greet")
    function greet(@Locale locale: string, @RequestParam name: string): Response {
        // @Locale 自动从 Accept-Language header 或 ?lang= 参数提取
        const msg = i18n.get("greeting", locale, Map.of(["name", name]))
        return Response.ok(Map.of(["message", msg]))
    }
}
```

## Locale 解析优先级

```
1. URL 参数:       ?lang=zh
2. Cookie:         lang=zh
3. HTTP Header:    Accept-Language: zh-CN,zh;q=0.9
4. 默认配置:       application.yml 中的 i18n.defaultLocale
```

## 配置

```yaml
# application.yml
i18n:
  defaultLocale: en
  supportedLocales:
    - en
    - zh
    - ja
  fallback: en        # 找不到翻译时回退到英文
  basePath: resources/i18n
```
